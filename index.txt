  <div id="kpi" style="margin:10px 0 14px; font-weight:700;"></div>

  <div class="row">

    <button id="downloadRadar">Скачать PNG (Колесо баланса)</button>

    <h3>Сферы: Факт vs Цель</h3>

    <h3>Radar chart (факт/цель, 0..1)</h3>

  </div>

  <div class="card">
    <h3>История ошибок API (локально)</h3>
    <div id="errorLog" style="font-size:14px;color:#c62828;">Ошибок пока нет</div>
  </div>

  <script>
    const API_URL = "https://script.google.com/macros/s/AKfycbzIPG8td7N6fQlQt9c_TY2p8bPtQStM4AMj5Jnm_vozrSxx_gW9rcsJYqnbUZEqb7yf/exec";

    let barChart = null;
    let radarChart = null;

    function getProgressColor(p) {
      if (p >= 100) return "#2e7d32"; // green
      if (p >= 50) return "#f9a825";  // yellow
      return "#c62828";               // red
    }

    function saveErrorLog(msg) {
      const key = "life_dashboard_api_errors";
      const arr = JSON.parse(localStorage.getItem(key) || "[]");
      arr.unshift(`${new Date().toLocaleString()} — ${msg}`);
      localStorage.setItem(key, JSON.stringify(arr.slice(0, 10)));
      renderErrorLog();
    }

    function renderErrorLog() {
      const key = "life_dashboard_api_errors";
      const arr = JSON.parse(localStorage.getItem(key) || "[]");
      const box = document.getElementById("errorLog");
      box.innerHTML = arr.length ? arr.map(x => `• ${x}`).join("<br>") : "Ошибок пока нет";
    }

    function saveLastSuccess() {
      const ts = new Date().toLocaleString();
      localStorage.setItem("life_dashboard_last_success", ts);
      document.getElementById("lastSuccess").textContent = `Последнее успешное обновление: ${ts}`;
    }

    function renderLastSuccess() {
      const ts = localStorage.getItem("life_dashboard_last_success");
      document.getElementById("lastSuccess").textContent = ts
        ? `Последнее успешное обновление: ${ts}`
        : "";
    }

    function renderProgress(rows) {
      const totalFact = rows.reduce((s, r) => s + (Number(r.fact) || 0), 0);
      const totalGoal = rows.reduce((s, r) => s + (Number(r.goal) || 0), 0);
      const totalPercent = totalGoal > 0 ? Math.round((totalFact / totalGoal) * 100) : 0;

      const kpi = document.getElementById("kpi");
      kpi.textContent = `Общий прогресс: ${totalFact}/${totalGoal} (${totalPercent}%)`;
      kpi.style.color = getProgressColor(totalPercent);

      const list = document.getElementById("progressList");
      list.innerHTML = rows.map(r => {
        const p = Number(r.progress) || 0;
        const color = getProgressColor(p);
        return `
          <div style="border:1px solid ${color};color:${color};background:#fff;border-radius:999px;padding:6px 10px;font-size:13px;">
            ${r.sphere}: ${r.fact}/${r.goal} (${p}%)
          </div>
        `;
      }).join("");
    }

    function renderStreaks(streaksObj) {
      const line = Object.entries(streaksObj)
        .map(([name, days]) => `${name}: ${days}`)
        .join(" | ");
      document.getElementById("streaks").textContent = `Дней подряд: ${line}`;
    }

    function downloadCanvas(canvasId, fileName) {
      const canvas = document.getElementById(canvasId);
      const link = document.createElement("a");
      link.href = canvas.toDataURL("image/png");
      link.download = fileName;
      link.click();
    }

    async function loadData() {
      try {
        const res = await fetch(API_URL, { cache: "no-store" });
        const data = await res.json();

        if (!data.ok || !Array.isArray(data.summary)) {
          throw new Error("Некорректный формат API");
        }

        const rows = data.summary;
        const streaks = data.streaks || {};

        renderProgress(rows);
        renderStreaks(streaks);

        const labels = rows.map(r => r.sphere);
        const fact = rows.map(r => Number(r.fact) || 0);
        const goal = rows.map(r => Number(r.goal) || 0);
        const progress = rows.map(r => Number(r.progress) || 0);

        // Bar
        if (barChart) barChart.destroy();
        barChart = new Chart(document.getElementById("barChart"), {
          type: "bar",
          data: {
            labels,
            datasets: [
              { label: "Факт", data: fact, backgroundColor: "#4e79a7" },
              { label: "Цель", data: goal, backgroundColor: "#f28e2b" }
            ]
          },
          options: {
            responsive: true,
            plugins: {
              tooltip: {
                callbacks: {
                  afterBody: (items) => `Выполнение: ${progress[items[0].dataIndex]}%`
                }
              }
            }
          }
        });

        // Radar (0..1)
        const ratio = rows.map(r => {
          const f = Number(r.fact) || 0;
          const g = Number(r.goal) || 0;
          return g > 0 ? Math.min(f / g, 1) : 0;
        });

        if (radarChart) radarChart.destroy();
        radarChart = new Chart(document.getElementById("radarChart"), {
          type: "radar",
          data: {
            labels,
            datasets: [{
              label: "Факт/Цель",
              data: ratio,
              fill: true,
              backgroundColor: "rgba(78,121,167,0.2)",
              borderColor: "#4e79a7",
              pointBackgroundColor: "#4e79a7"
            }]
          },
          options: {
            responsive: true,
            scales: {
              r: {
                min: 0,
                max: 1,
                ticks: { stepSize: 0.2 }
              }
            }
          }
        });

        document.getElementById("status").textContent = `Обновлено: ${new Date().toLocaleTimeString()}`;
        saveLastSuccess();
      } catch (err) {
        document.getElementById("status").textContent = "Ошибка API: " + err.message;
        saveErrorLog(err.message);
      }
    }

    document.getElementById("downloadBar").addEventListener("click", () => {
      downloadCanvas("barChart", "fact-vs-goal.png");
    });

    document.getElementById("downloadRadar").addEventListener("click", () => {
      downloadCanvas("radarChart", "balance-wheel.png");
    });

    renderErrorLog();
    renderLastSuccess();
    loadData();
    setInterval(loadData, 30000);
  </script>
